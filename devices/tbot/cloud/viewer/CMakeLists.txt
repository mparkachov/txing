cmake_minimum_required(VERSION 3.21)
project(txing_tbot_kvs_viewer LANGUAGES C CXX)
include(CTest)
if(APPLE)
  execute_process(COMMAND brew --prefix OUTPUT_VARIABLE _brew OUTPUT_STRIP_TRAILING_WHITESPACE)
  list(APPEND CMAKE_PREFIX_PATH "${_brew}" "${_brew}/opt/openssl@3")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../common/kvs/cmake/AwsKvsWebRtc.cmake")
add_executable(txing-tbot-kvs-viewer main.cpp)
target_compile_features(txing-tbot-kvs-viewer PRIVATE cxx_std_17)
target_compile_options(txing-tbot-kvs-viewer PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(txing-tbot-kvs-viewer PRIVATE
  TXING_KVS_SYSTEM_CA_CERT_PATH="${TXING_KVS_SYSTEM_CA_CERT_PATH}")
txing_link_kvs_sdk(txing-tbot-kvs-viewer)
install(TARGETS txing-tbot-kvs-viewer RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  add_executable(txing-tbot-kvs-viewer-peer-test peer_test.cpp loopback_interfaces.c)
  target_compile_features(txing-tbot-kvs-viewer-peer-test PRIVATE cxx_std_17)
  target_include_directories(txing-tbot-kvs-viewer-peer-test PRIVATE "${TXING_AWS_KVS_WEBRTC_SDK_INSTALL_DIR}/include")
  get_target_property(_sdk_libraries txing-tbot-kvs-viewer LINK_LIBRARIES)
  target_link_libraries(txing-tbot-kvs-viewer-peer-test PRIVATE ${_sdk_libraries})
  add_dependencies(txing-tbot-kvs-viewer-peer-test txing_aws_kvs_webrtc_sdk)
  add_test(NAME real-kvs-viewer-peers COMMAND txing-tbot-kvs-viewer-peer-test)
  set_tests_properties(real-kvs-viewer-peers PROPERTIES TIMEOUT 45)
endif()
